\section{Introduction}
% ============================================================
\begin{frame}{Index}
\tableofcontents[currentsection]
\end{frame}

\begin{frame}{Being born is the single most important lottery you play}
\begin{itemize}
    \item Location at birth is largely a roll of the dice
    \item Rolling bad on climate yields a lifelong disamenity
    \item Improving local climate is extremely costly
    \item Migration is thus the main escape mechanism
\end{itemize}
\vspace{0.5em}
\begin{block}{Key Question}
Do climate disamenities make people respond more strongly to reductions in migration costs?
\end{block}
\vspace{0.3em}
\begin{center}
And if so, \textbf{does this drive regional economic development?}
\end{center}
\end{frame}


\begin{frame}{Contribution}
Climate migration events and their consequences have been studied far and wide
\begin{itemize}
    \item Disasters may lead to permanent migration (Belasen \& Polacheck 2013) or to temporary flows (Stanley 2010)
    \item Plain bad climate may also drive moves (Saldana 2009; Halliday 2006)
    \item Model implied welfare measures point to reasonable-to-large gains (Desmet \& Rossi-Hansberg 2020; Morten \& Oliveira 2025; Kennan \& Walker 2008)
    \item As do measurements of individual level income gains (Bryan \& Mobarak 2014; Hornbeck 2002)
\end{itemize}
\textbf{This project:} Are migrations responses to costs also amplified? And, can we get a more direct aggregate welfare measure?
\par
Laterally, but important to me: better understanding regional development stories in Brazil
\end{frame}


\begin{frame}{Findings}
The main preliminary results are that:
\begin{itemize}
    \item \textbf{Climate:} No robust effect on migration responses
    \item \textbf{Welfare:} Migration leads to positive, sizable GDP per capita gains
    \begin{itemize}
        \item I find a 2.3\% increase in GDP per capita attributable to migration
        \item The closest analogue, Morten-Oliveira, find a 0.6\% total welfare effect
    \end{itemize}
    \item Both results seem robust 
\end{itemize}

\vspace{0.5em}
\textbf{Caveat:} Data limits the amount of controls available for the welfare result
\end{frame}

\section{Identification}
% ============================================================
\begin{frame}{Index}
\tableofcontents[currentsection]
\end{frame}

\begin{frame}{Central Identification Idea}
I leverage the Morten-Oliveira instrument to obtain an exogenous shock to travel times
\textbf{Climate:}
\begin{itemize}
    \item Use transport shocks to identify migration responses
    \item Caveat: FEs capture region specific effects
    \begin{itemize}
        \item Test: do people in high-exposure regions respond more strongly to the opportunity to leave?
    \end{itemize}
\end{itemize}
\textbf{Welfare:}
\begin{itemize}
    \item Recover migration induced by transport shocks
    \item This should be exogenous to regional welfare outcomes
    \begin{itemize}
        \item Test: does net migration drive local welfare?
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{The welfare channel}
GDP per capita is a good measure of welfare, but is codetermined with migration flows \par 
$\Rightarrow$  the Brasília instrument allows for exogenous variation of migration!

\vspace{0.5em}
\textbf{Key issue: controls}
\begin{itemize}
    \item Exogeneity does not mean absence of omitted variables
    \item Road construction itself could drive GDP per capita outcomes
    \item I implement a gravity theory inspired market access control
\end{itemize}
\end{frame}

\section{Further Work}
% ============================================================
\begin{frame}{Index}
\tableofcontents[currentsection]
\end{frame}

\begin{frame}{Robustness Checks}
I'm satisfied with the climate null result. 
A better defense of the GDP/Welfare result requires:
\begin{itemize}
    \item Replication entire pipeline at state level
    \item Better controls for omitted variables
\end{itemize}
The current setup allows for exogenous migration, but roads impact GDP in many other ways
\end{frame}


\begin{frame}{Interpretation}
Provided the GDP result is convincing, there are many ways to explore what drives it
\begin{itemize}
    \item Selection: census microdata comparison of types of migrants
    \item Agglomeration: move away from the buzzword. Why do cities feature agglomeration and other places do not?
    \item Human capital accumulation? Better matching?
    \item What about the agricultural frontier? Resource abundance? A model could be useful here.
\end{itemize}
Figuring this out could go a long way towards building a better understanding of Brazilian mid-to-late 20th century development!
\end{frame}
